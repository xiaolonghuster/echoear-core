cmake_minimum_required(VERSION 3.16)
if(ESP_PLATFORM)
    idf_component_register(SRCS "src/original_icons.cc" INCLUDE_DIRS "include" REQUIRES lvgl__lvgl)
else()
    project(echoear_core LANGUAGES CXX)
    enable_testing()
    add_library(echoear_core INTERFACE)
    target_include_directories(echoear_core INTERFACE include)
    target_compile_features(echoear_core INTERFACE cxx_std_17)
    file(GLOB ECHOEAR_TESTS "tests/test_*.cc")
    foreach(source IN LISTS ECHOEAR_TESTS)
        get_filename_component(name "${source}" NAME_WE)
        add_executable(${name} "${source}")
        target_link_libraries(${name} PRIVATE echoear_core)
        add_test(NAME ${name} COMMAND ${name})
    endforeach()
endif()
